# ==================== FILE: bookbridge/security.py ====================
import base64
import hashlib
import hmac
import os

PBKDF2_ITERATIONS = 240_000


def hash_password(password: str) -> str:
    if not isinstance(password, str) or len(password) < 8:
        raise ValueError("Password must contain at least 8 characters")
    salt = os.urandom(16)
    digest = hashlib.pbkdf2_hmac(
        "sha256",
        password.encode("utf-8"),
        salt,
        PBKDF2_ITERATIONS,
    )
    return "pbkdf2_sha256${}${}${}".format(
        PBKDF2_ITERATIONS,
        base64.b64encode(salt).decode("ascii"),
        base64.b64encode(digest).decode("ascii"),
    )


def verify_password(password: str, encoded: str) -> bool:
    try:
        algorithm, iterations, salt_b64, digest_b64 = encoded.split("$", 3)
        if algorithm != "pbkdf2_sha256":
            return False
        salt = base64.b64decode(salt_b64)
        expected = base64.b64decode(digest_b64)
        actual = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            int(iterations),
        )
        return hmac.compare_digest(actual, expected)
    except (AttributeError, ValueError, TypeError):
        return False

from __future__ import annotations

SCHEMA = r"""
PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS users (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    full_name TEXT NOT NULL,
    email TEXT NOT NULL UNIQUE COLLATE NOCASE,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL CHECK(role IN ('buyer','seller','admin')),
    is_active INTEGER NOT NULL DEFAULT 1 CHECK(is_active IN (0,1)),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS products (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    seller_id INTEGER NOT NULL,
    title TEXT NOT NULL,
    author TEXT NOT NULL,
    isbn TEXT,
    category TEXT NOT NULL,
    description TEXT NOT NULL DEFAULT '',
    price_kobo INTEGER NOT NULL CHECK(price_kobo >= 0),
    stock INTEGER NOT NULL CHECK(stock >= 0),
    status TEXT NOT NULL DEFAULT 'active' CHECK(status IN ('active','inactive')),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(seller_id) REFERENCES users(id)
);

CREATE TABLE IF NOT EXISTS cart_items (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    buyer_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    quantity INTEGER NOT NULL CHECK(quantity > 0),
    UNIQUE(buyer_id, product_id),
    FOREIGN KEY(buyer_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY(product_id) REFERENCES products(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS orders (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    buyer_id INTEGER NOT NULL,
    total_kobo INTEGER NOT NULL CHECK(total_kobo >= 0),
    shipping_address TEXT NOT NULL,
    payment_method TEXT NOT NULL CHECK(payment_method IN ('bank_transfer','cash_on_delivery','demo_card')),
    payment_status TEXT NOT NULL CHECK(payment_status IN ('pending','paid','failed')),
    order_status TEXT NOT NULL CHECK(order_status IN ('pending','processing','shipped','delivered','cancelled')),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(buyer_id) REFERENCES users(id)
);

CREATE TABLE IF NOT EXISTS order_items (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    order_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    seller_id INTEGER NOT NULL,
    title_snapshot TEXT NOT NULL,
    quantity INTEGER NOT NULL CHECK(quantity > 0),
    unit_price_kobo INTEGER NOT NULL CHECK(unit_price_kobo >= 0),
    subtotal_kobo INTEGER NOT NULL CHECK(subtotal_kobo >= 0),
    fulfilment_status TEXT NOT NULL DEFAULT 'pending' CHECK(fulfilment_status IN ('pending','processing','shipped','delivered','cancelled')),
    FOREIGN KEY(order_id) REFERENCES orders(id) ON DELETE CASCADE,
    FOREIGN KEY(product_id) REFERENCES products(id),
    FOREIGN KEY(seller_id) REFERENCES users(id)
);

CREATE TABLE IF NOT EXISTS payments (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    order_id INTEGER NOT NULL UNIQUE,
    reference TEXT NOT NULL UNIQUE,
    amount_kobo INTEGER NOT NULL CHECK(amount_kobo >= 0),
    method TEXT NOT NULL CHECK(method IN ('bank_transfer','cash_on_delivery','demo_card')),
    status TEXT NOT NULL CHECK(status IN ('pending','paid','failed')),
    created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(order_id) REFERENCES orders(id) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS idx_products_search ON products(title, author, category);
CREATE INDEX IF NOT EXISTS idx_products_seller ON products(seller_id, status);
CREATE INDEX IF NOT EXISTS idx_orders_buyer ON orders(buyer_id, created_at);
CREATE INDEX IF NOT EXISTS idx_order_items_seller ON order_items(seller_id, fulfilment_status);
"""


class Database:
    def __init__(self, path: str | Path = "bookbridge.db") -> None:
        self.path = str(path)

    def connect(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self.path, timeout=30.0)
        conn.row_factory = sqlite3.Row
        conn.execute("PRAGMA foreign_keys = ON")
        conn.execute("PRAGMA busy_timeout = 30000")
        return conn

    def initialize(self, seed_demo: bool = True) -> None:
        with self.connect() as conn:
            conn.executescript(SCHEMA)
            if seed_demo:
                self._seed_demo(conn)

    @contextmanager
    def transaction(self) -> Iterator[sqlite3.Connection]:
        conn = self.connect()
        try:
            conn.execute("BEGIN IMMEDIATE")
            yield conn
            conn.commit()
        except Exception:
            conn.rollback()
            raise
        finally:
            conn.close()

from __future__ import annotations

import re
import secrets
import sqlite3
from dataclasses import dataclass
from decimal import Decimal, InvalidOperation, ROUND_HALF_UP
from typing import Any

from .database import Database
from .security import hash_password, verify_password

EMAIL_RE = re.compile(r"^[^\s@]+@[^\s@]+\.[^\s@]+$")
PUBLIC_ROLES = {"buyer", "seller"}
PRODUCT_STATUSES = {"active", "inactive"}
PAYMENT_METHODS = {"bank_transfer", "cash_on_delivery", "demo_card"}
FULFILMENT_STATUSES = {"pending", "processing", "shipped", "delivered", "cancelled"}


class MarketplaceError(Exception):
    pass


class ValidationError(MarketplaceError):
    pass


class AuthenticationError(MarketplaceError):
    pass


class AuthorizationError(MarketplaceError):
    pass


class NotFoundError(MarketplaceError):
    pass


@dataclass(frozen=True)
class SessionUser:
    id: int
    full_name: str
    email: str
    role: str


def money_to_kobo(value: Any) -> int:
    try:
        amount = Decimal(str(value)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)
    except (InvalidOperation, ValueError, TypeError):
        raise ValidationError("Enter a valid price")
    if amount < 0:
        raise ValidationError("Price cannot be negative")
    return int(amount * 100)


def kobo_to_naira(kobo: int) -> str:
    return f"₦{Decimal(kobo) / Decimal(100):,.2f}"


class MarketplaceService:
    def __init__(self, db: Database) -> None:
        self.db = db

    def register(self, full_name: str, email: str, password: str, role: str) -> int:
        full_name = (full_name or "").strip()
        email = (email or "").strip().lower()
        role = (role or "").strip().lower()

        if len(full_name) < 2:
            raise ValidationError("Enter your full name")
        if not EMAIL_RE.match(email):
            raise ValidationError("Enter a valid email address")
        if role not in PUBLIC_ROLES:
            raise ValidationError("Role must be buyer or seller")
        if len(password or "") < 8:
            raise ValidationError("Password must contain at least 8 characters")

        try:
            encoded = hash_password(password)
            with self.db.connect() as conn:
                cur = conn.execute(
                    "INSERT INTO users(full_name,email,password_hash,role) VALUES (?,?,?,?)",
                    (full_name, email, encoded, role),
                )
                return int(cur.lastrowid)
        except sqlite3.IntegrityError as exc:
            if "users.email" in str(exc).lower() or "unique" in str(exc).lower():
                raise ValidationError("An account with that email already exists") from exc
            raise

    def login(self, email: str, password: str) -> SessionUser:
        email = (email or "").strip().lower()
        with self.db.connect() as conn:
            row = conn.execute(
                "SELECT * FROM users WHERE email=? AND is_active=1",
                (email,),
            ).fetchone()
        if not row or not verify_password(password or "", row["password_hash"]):
            raise AuthenticationError("Invalid email or password")
        return SessionUser(
            id=int(row["id"]),
            full_name=row["full_name"],
            email=row["email"],
            role=row["role"],
        )

    def save_product(
        self,
        actor: SessionUser,
        title: str,
        author: str,
        isbn: str,
        category: str,
        description: str,
        price_naira: Any,
        stock: Any,
        product_id: int | None = None,
        status: str = "active",
    ) -> int:
        if actor.role not in {"seller", "admin"}:
            raise AuthorizationError("Seller or administrator access is required")

        title = (title or "").strip()
        author = (author or "").strip()
        isbn = (isbn or "").strip()
        category = (category or "").strip()
        description = (description or "").strip()
        status = (status or "active").strip().lower()

        if not title or not author or not category:
            raise ValidationError("Title, author and category are required")
        if status not in PRODUCT_STATUSES:
            raise ValidationError("Invalid listing status")
        price_kobo = money_to_kobo(price_naira)
        try:
            stock_int = int(stock)
        except (TypeError, ValueError):
            raise ValidationError("Stock must be a whole number")
        if stock_int < 0:
            raise ValidationError("Stock cannot be negative")

        with self.db.connect() as conn:
            if product_id is None:
                if actor.role == "admin":
                    raise ValidationError("Administrator must specify an existing seller-owned listing")
                cur = conn.execute(
                    """
                    INSERT INTO products(
                        seller_id,title,author,isbn,category,description,price_kobo,stock,status
                    ) VALUES (?,?,?,?,?,?,?,?,?)
                    """,
                    (
                        actor.id,
                        title,
                        author,
                        isbn,
                        category,
                        description,
                        price_kobo,
                        stock_int,
                        status,
                    ),
                )
                return int(cur.lastrowid)

            row = conn.execute(
                "SELECT seller_id FROM products WHERE id=?",
                (product_id,),
            ).fetchone()
            if not row:
                raise NotFoundError("Product not found")
            if actor.role != "admin" and int(row["seller_id"]) != actor.id:
                raise AuthorizationError("You can update only your own listings")
            conn.execute(
                """
                UPDATE products
                SET title=?,author=?,isbn=?,category=?,description=?,price_kobo=?,stock=?,status=?,updated_at=CURRENT_TIMESTAMP
                WHERE id=?
                """,
                (
                    title,
                    author,
                    isbn,
                    category,
                    description,
                    price_kobo,
                    stock_int,
                    status,
                    product_id,
                ),
            )
            return int(product_id)

    def set_product_status(self, actor: SessionUser, product_id: int, status: str) -> None:
        status = (status or "").strip().lower()
        if status not in PRODUCT_STATUSES:
            raise ValidationError("Invalid listing status")
        with self.db.connect() as conn:
            row = conn.execute("SELECT seller_id FROM products WHERE id=?", (product_id,)).fetchone()
            if not row:
                raise NotFoundError("Product not found")
            if actor.role != "admin" and not (actor.role == "seller" and int(row["seller_id"]) == actor.id):
                raise AuthorizationError("Not permitted to change this listing")
            conn.execute(
                "UPDATE products SET status=?,updated_at=CURRENT_TIMESTAMP WHERE id=?",
                (status, product_id),
            )

    def seller_products(self, actor: SessionUser) -> list[dict[str, Any]]:
        if actor.role not in {"seller", "admin"}:
            raise AuthorizationError("Seller or administrator access is required")
        with self.db.connect() as conn:
            if actor.role == "admin":
                rows = conn.execute(
                    """
                    SELECT p.*,u.full_name AS seller_name,u.email AS seller_email
                    FROM products p JOIN users u ON u.id=p.seller_id
                    ORDER BY p.created_at DESC,p.id DESC
                    """
                ).fetchall()
            else:
                rows = conn.execute(
                    "SELECT * FROM products WHERE seller_id=? ORDER BY created_at DESC,id DESC",
                    (actor.id,),
                ).fetchall()
        return [dict(r) for r in rows]

    def categories(self) -> list[str]:
        with self.db.connect() as conn:
            rows = conn.execute(
                "SELECT DISTINCT category FROM products WHERE status='active' ORDER BY category COLLATE NOCASE"
            ).fetchall()
        return [r["category"] for r in rows]

    def list_products(self, query: str = "", category: str = "") -> list[dict[str, Any]]:
        query = (query or "").strip()
        category = (category or "").strip()
        params: list[Any] = []
        where = ["p.status='active'", "p.stock>0", "u.is_active=1"]
        if query:
            pattern = f"%{query}%"
            where.append(
                "(p.title LIKE ? COLLATE NOCASE OR p.author LIKE ? COLLATE NOCASE OR p.isbn LIKE ? COLLATE NOCASE OR p.description LIKE ? COLLATE NOCASE)"
            )
            params.extend([pattern, pattern, pattern, pattern])
        if category:
            where.append("p.category=? COLLATE NOCASE")
            params.append(category)
        sql = f"""
            SELECT p.*,u.full_name AS seller_name
            FROM products p JOIN users u ON u.id=p.seller_id
            WHERE {' AND '.join(where)}
            ORDER BY p.title COLLATE NOCASE,p.id DESC
        """
        with self.db.connect() as conn:
            rows = conn.execute(sql, params).fetchall()
        return [dict(r) for r in rows]

    def add_to_cart(self, buyer: SessionUser, product_id: int, quantity: int = 1) -> None:
        self._require_role(buyer, "buyer")
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValidationError("Quantity must be a whole number")
        if quantity <= 0:
            raise ValidationError("Quantity must be greater than zero")

        with self.db.transaction() as conn:
            product = conn.execute(
                "SELECT id,stock,status FROM products WHERE id=?",
                (product_id,),
            ).fetchone()
            if not product or product["status"] != "active" or int(product["stock"]) <= 0:
                raise ValidationError("This product is unavailable")
            existing = conn.execute(
                "SELECT quantity FROM cart_items WHERE buyer_id=? AND product_id=?",
                (buyer.id, product_id),
            ).fetchone()
            new_quantity = quantity + (int(existing["quantity"]) if existing else 0)
            if new_quantity > int(product["stock"]):
                raise ValidationError("Requested quantity exceeds available stock")
            conn.execute(
                """
                INSERT INTO cart_items(buyer_id,product_id,quantity)
                VALUES (?,?,?)
                ON CONFLICT(buyer_id,product_id)
                DO UPDATE SET quantity=excluded.quantity
                """,
                (buyer.id, product_id, new_quantity),
            )

    def set_cart_quantity(self, buyer: SessionUser, product_id: int, quantity: int) -> None:
        self._require_role(buyer, "buyer")
        try:
            quantity = int(quantity)
        except (TypeError, ValueError):
            raise ValidationError("Quantity must be a whole number")
        if quantity <= 0:
            self.remove_from_cart(buyer, product_id)
            return
        with self.db.transaction() as conn:
            product = conn.execute(
                "SELECT stock,status FROM products WHERE id=?",
                (product_id,),
            ).fetchone()
            if not product or product["status"] != "active":
                raise ValidationError("This product is unavailable")
            if quantity > int(product["stock"]):
                raise ValidationError("Requested quantity exceeds available stock")
            cur = conn.execute(
                "UPDATE cart_items SET quantity=? WHERE buyer_id=? AND product_id=?",
                (quantity, buyer.id, product_id),
            )
            if cur.rowcount == 0:
                raise NotFoundError("Cart item not found")

    def remove_from_cart(self, buyer: SessionUser, product_id: int) -> None:
        self._require_role(buyer, "buyer")
        with self.db.connect() as conn:
            conn.execute(
                "DELETE FROM cart_items WHERE buyer_id=? AND product_id=?",
                (buyer.id, product_id),
            )

    def cart(self, buyer_id: int) -> list[dict[str, Any]]:
        with self.db.connect() as conn:
            rows = conn.execute(
                """
                SELECT c.product_id,c.quantity,p.title,p.author,p.price_kobo,p.stock,p.status,
                       u.full_name AS seller_name,
                       (c.quantity*p.price_kobo) AS subtotal_kobo
                FROM cart_items c
                JOIN products p ON p.id=c.product_id
                JOIN users u ON u.id=p.seller_id
                WHERE c.buyer_id=?
                ORDER BY p.title COLLATE NOCASE
                """,
                (buyer_id,),
            ).fetchall()
        return [dict(r) for r in rows]

    def checkout(self, buyer: SessionUser, shipping_address: str, payment_method: str) -> int:
        self._require_role(buyer, "buyer")
        shipping_address = (shipping_address or "").strip()
        payment_method = (payment_method or "").strip().lower()
        if len(shipping_address) < 8:
            raise ValidationError("Enter a complete delivery address")
        if payment_method not in PAYMENT_METHODS:
            raise ValidationError("Select a valid payment method")

        with self.db.transaction() as conn:
            rows = conn.execute(
                """
                SELECT c.product_id,c.quantity,p.title,p.seller_id,p.price_kobo,p.stock,p.status
                FROM cart_items c
                JOIN products p ON p.id=c.product_id
                WHERE c.buyer_id=?
                """,
                (buyer.id,),
            ).fetchall()
            if not rows:
                raise ValidationError("Your cart is empty")

            total = 0
            for row in rows:
                if row["status"] != "active" or int(row["quantity"]) > int(row["stock"]):
                    raise ValidationError(f"Insufficient stock for {row['title']}")
                total += int(row["quantity"]) * int(row["price_kobo"])

            payment_status = "paid" if payment_method == "demo_card" else "pending"
            order_status = "processing" if payment_status == "paid" else "pending"
            cur = conn.execute(
                """
                INSERT INTO orders(
                    buyer_id,total_kobo,shipping_address,payment_method,payment_status,order_status
                ) VALUES (?,?,?,?,?,?)
                """,
                (buyer.id, total, shipping_address, payment_method, payment_status, order_status),
            )
            order_id = int(cur.lastrowid)

            for row in rows:
                qty = int(row["quantity"])
                price = int(row["price_kobo"])
                subtotal = qty * price
                conn.execute(
                    """
                    INSERT INTO order_items(
                        order_id,product_id,seller_id,title_snapshot,quantity,unit_price_kobo,subtotal_kobo
                    ) VALUES (?,?,?,?,?,?,?)
                    """,
                    (
                        order_id,
                        row["product_id"],
                        row["seller_id"],
                        row["title"],
                        qty,
                        price,
                        subtotal,
                    ),
                )
                updated = conn.execute(
                    "UPDATE products SET stock=stock-?,updated_at=CURRENT_TIMESTAMP WHERE id=? AND status='active' AND stock>=?",
                    (qty, row["product_id"], qty),
                )
                if updated.rowcount != 1:
                    raise ValidationError(f"Stock changed for {row['title']}; please review your cart")

            reference = "BB-" + secrets.token_hex(6).upper()
            conn.execute(
                "INSERT INTO payments(order_id,reference,amount_kobo,method,status) VALUES (?,?,?,?,?)",
                (order_id, reference, total, payment_method, payment_status),
            )
            conn.execute("DELETE FROM cart_items WHERE buyer_id=?", (buyer.id,))
            return order_id

    def buyer_orders(self, buyer_id: int) -> list[dict[str, Any]]:
        with self.db.connect() as conn:
            orders = conn.execute(
                """
                SELECT o.*,p.reference AS payment_reference
                FROM orders o
                JOIN payments p ON p.order_id=o.id
                WHERE o.buyer_id=?
                ORDER BY o.created_at DESC,o.id DESC
                """,
                (buyer_id,),
            ).fetchall()
            result: list[dict[str, Any]] = []
            for order in orders:
                payload = dict(order)
                items = conn.execute(
                    """
                    SELECT oi.*,u.full_name AS seller_name
                    FROM order_items oi JOIN users u ON u.id=oi.seller_id
                    WHERE oi.order_id=? ORDER BY oi.id
                    """,
                    (order["id"],),
                ).fetchall()
                payload["items"] = [dict(i) for i in items]
                result.append(payload)
        return result

    def seller_orders(self, seller_id: int) -> list[dict[str, Any]]:
        with self.db.connect() as conn:
            rows = conn.execute(
                """
                SELECT oi.*,o.created_at,o.shipping_address,o.payment_method,o.payment_status,o.order_status,
                       b.full_name AS buyer_name,b.email AS buyer_email
                FROM order_items oi
                JOIN orders o ON o.id=oi.order_id
                JOIN users b ON b.id=o.buyer_id
                WHERE oi.seller_id=?
                ORDER BY o.created_at DESC,oi.id DESC
                """,
                (seller_id,),
            ).fetchall()
        return [dict(r) for r in rows]

    def update_fulfilment(self, actor: SessionUser, order_item_id: int, status: str) -> None:
        status = (status or "").strip().lower()
        if status not in FULFILMENT_STATUSES:
            raise ValidationError("Invalid fulfilment status")
        if actor.role not in {"seller", "admin"}:
            raise AuthorizationError("Seller or administrator access is required")

        with self.db.transaction() as conn:
            item = conn.execute(
                "SELECT order_id,seller_id FROM order_items WHERE id=?",
                (order_item_id,),
            ).fetchone()
            if not item:
                raise NotFoundError("Order item not found")
            if actor.role != "admin" and int(item["seller_id"]) != actor.id:
                raise AuthorizationError("You can update only your own order items")

            conn.execute(
                "UPDATE order_items SET fulfilment_status=? WHERE id=?",
                (status, order_item_id),
            )
            statuses = [
                r["fulfilment_status"]
                for r in conn.execute(
                    "SELECT fulfilment_status FROM order_items WHERE order_id=?",
                    (item["order_id"],),
                ).fetchall()
            ]
            order_status = self._derive_order_status(statuses)
            conn.execute(
                "UPDATE orders SET order_status=? WHERE id=?",
                (order_status, item["order_id"]),
            )

    def admin_metrics(self, actor: SessionUser) -> dict[str, Any]:
        self._require_role(actor, "admin")
        with self.db.connect() as conn:
            users = conn.execute("SELECT COUNT(*) AS c FROM users").fetchone()["c"]
            buyers = conn.execute("SELECT COUNT(*) AS c FROM users WHERE role='buyer'").fetchone()["c"]
            sellers = conn.execute("SELECT COUNT(*) AS c FROM users WHERE role='seller'").fetchone()["c"]
            products = conn.execute("SELECT COUNT(*) AS c FROM products").fetchone()["c"]
            active_products = conn.execute("SELECT COUNT(*) AS c FROM products WHERE status='active'").fetchone()["c"]
            orders = conn.execute("SELECT COUNT(*) AS c FROM orders").fetchone()["c"]
            paid_revenue = conn.execute(
                "SELECT COALESCE(SUM(amount_kobo),0) AS total FROM payments WHERE status='paid'"
            ).fetchone()["total"]
            pending_payments = conn.execute(
                "SELECT COUNT(*) AS c FROM payments WHERE status='pending'"
            ).fetchone()["c"]
        return {
            "users": int(users),
            "buyers": int(buyers),
            "sellers": int(sellers),
            "products": int(products),
            "active_products": int(active_products),
            "orders": int(orders),
            "paid_revenue_kobo": int(paid_revenue),
            "pending_payments": int(pending_payments),
        }

    def set_user_active(self, actor: SessionUser, user_id: int, is_active: bool) -> None:
        self._require_role(actor, "admin")
        if user_id == actor.id and not is_active:
            raise ValidationError("Administrator cannot deactivate the current session")
        with self.db.connect() as conn:
            cur = conn.execute(
                "UPDATE users SET is_active=? WHERE id=?",
                (1 if is_active else 0, user_id),
            )
            if cur.rowcount == 0:
                raise NotFoundError("User not found")

    def admin_users(self, actor: SessionUser) -> list[dict[str, Any]]:
        self._require_role(actor, "admin")
        with self.db.connect() as conn:
            rows = conn.execute(
                "SELECT id,full_name,email,role,is_active,created_at FROM users ORDER BY created_at DESC,id DESC"
            ).fetchall()
        return [dict(r) for r in rows]

    @staticmethod
    def _derive_order_status(statuses: list[str]) -> str:
        if not statuses:
            return "pending"
        if all(s == "cancelled" for s in statuses):
            return "cancelled"
        active = [s for s in statuses if s != "cancelled"]
        if active and all(s == "delivered" for s in active):
            return "delivered"
        if any(s == "shipped" for s in active):
            return "shipped"
        if any(s in {"processing", "delivered"} for s in active):
            return "processing"
        return "pending"

    @staticmethod
    def _require_role(actor: SessionUser, role: str) -> None:
        if actor.role != role:
            raise AuthorizationError(f"{role.capitalize()} access is required")

class MarketplaceServiceTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.db = Database(Path(self.tmp.name) / "test.db")
        self.db.initialize(seed_demo=False)
        self.svc = MarketplaceService(self.db)
        self.buyer_id = self.svc.register("Buyer One", "buyer@example.com", "Buyer123!", "buyer")
        self.seller_id = self.svc.register("Seller One", "seller@example.com", "Seller123!", "seller")
        self.other_seller_id = self.svc.register("Seller Two", "seller2@example.com", "Seller223!", "seller")
        self.buyer = self.svc.login("buyer@example.com", "Buyer123!")
        self.seller = self.svc.login("seller@example.com", "Seller123!")
        self.other_seller = self.svc.login("seller2@example.com", "Seller223!")
        self.product_id = self.svc.save_product(
            self.seller,
            "Software Engineering",
            "Ian Sommerville",
            "9780137035151",
            "Computing",
            "Software engineering text",
            12000.00,
            4,
        )

    def tearDown(self):
        self.tmp.cleanup()

    def test_registration_and_login(self):
        user = self.svc.login("buyer@example.com", "Buyer123!")
        self.assertEqual(user.role, "buyer")

    def test_duplicate_registration_is_rejected(self):
        with self.assertRaises(ValidationError):
            self.svc.register("Another", "buyer@example.com", "Password1!", "buyer")

    def test_invalid_login_is_rejected(self):
        with self.assertRaises(AuthenticationError):
            self.svc.login("buyer@example.com", "wrong-password")

    def test_search_finds_product_by_author(self):
        products = self.svc.list_products("Sommerville")
        self.assertEqual(len(products), 1)
        self.assertEqual(products[0]["id"], self.product_id)

    def test_category_filter(self):
        self.assertEqual(len(self.svc.list_products("", "Computing")), 1)
        self.assertEqual(len(self.svc.list_products("", "Literature")), 0)

    def test_cart_rejects_quantity_above_stock(self):
        with self.assertRaises(ValidationError):
            self.svc.add_to_cart(self.buyer, self.product_id, 5)

    def test_cart_quantity_can_be_updated_and_removed(self):
        self.svc.add_to_cart(self.buyer, self.product_id, 1)
        self.svc.set_cart_quantity(self.buyer, self.product_id, 3)
        self.assertEqual(self.svc.cart(self.buyer.id)[0]["quantity"], 3)
        self.svc.remove_from_cart(self.buyer, self.product_id)
        self.assertEqual(self.svc.cart(self.buyer.id), [])

    def test_buyer_cannot_create_product(self):
        with self.assertRaises(AuthorizationError):
            self.svc.save_product(self.buyer, "X", "Y", "", "Computing", "", 10, 1)

    def test_seller_cannot_edit_another_sellers_product(self):
        with self.assertRaises(AuthorizationError):
            self.svc.save_product(
                self.other_seller,
                "Changed",
                "Author",
                "",
                "Computing",
                "",
                100,
                1,
                product_id=self.product_id,
            )

    def test_checkout_creates_order_and_reduces_stock(self):
        self.svc.add_to_cart(self.buyer, self.product_id, 2)
        order_id = self.svc.checkout(self.buyer, "12 University Road, Yola", "demo_card")
        orders = self.svc.buyer_orders(self.buyer.id)
        self.assertEqual(orders[0]["id"], order_id)
        self.assertEqual(orders[0]["payment_status"], "paid")
        products = self.svc.list_products("Software Engineering")
        self.assertEqual(products[0]["stock"], 2)
        self.assertEqual(self.svc.cart(self.buyer.id), [])

    def test_empty_cart_checkout_is_rejected(self):
        with self.assertRaises(ValidationError):
            self.svc.checkout(self.buyer, "12 University Road, Yola", "demo_card")

    def test_seller_updates_fulfilment(self):
        self.svc.add_to_cart(self.buyer, self.product_id, 1)
        self.svc.checkout(self.buyer, "12 University Road, Yola", "demo_card")
        seller_items = self.svc.seller_orders(self.seller.id)
        self.svc.update_fulfilment(self.seller, seller_items[0]["id"], "shipped")
        seller_items = self.svc.seller_orders(self.seller.id)
        self.assertEqual(seller_items[0]["fulfilment_status"], "shipped")
        self.assertEqual(self.svc.buyer_orders(self.buyer.id)[0]["order_status"], "shipped")

    def test_other_seller_cannot_update_fulfilment(self):
        self.svc.add_to_cart(self.buyer, self.product_id, 1)
        self.svc.checkout(self.buyer, "12 University Road, Yola", "demo_card")
        item_id = self.svc.seller_orders(self.seller.id)[0]["id"]
        with self.assertRaises(AuthorizationError):
            self.svc.update_fulfilment(self.other_seller, item_id, "delivered")


if __name__ == "__main__":
    unittest.main()
